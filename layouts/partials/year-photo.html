{{- /* The photo on a year's tile on /archive/ (Round 4b). Camera photos (a JPEG or HEIC) come first, since screenshots and graphics are PNGs: the year's first Featured post with one, else the first from July on, else the year's last. Then the same order with any photo, else "" (2000 has none). Takes the year's section page; call with partialCached keyed by the year. params.featured_category names the category (default "Featured"). */ -}} {{- $typeName := site.Params.contentTypeName | default "post" -}} {{- $pages := (where (where .RegularPagesRecursive "Type" $typeName) "Params.photos" "!=" nil).ByDate -}} {{- $featured := where $pages "Params.categories" "intersect" (slice (site.Params.featured_category | default "Featured")) -}} {{- $later := where $pages "Date" "ge" (time.AsTime (printf "%s-07-01T00:00:00Z" .Section)) -}} {{- $out := "" -}} {{- range $pattern := slice `(?i)\.(jpe?g|heic)(\?.*)?$` "" }} {{- range $list := slice $featured $later $pages.Reverse }} {{- range $list }} {{- with partial "first-image.html" . }}{{ if or (not $pattern) (findRE $pattern . 1) }}{{ $out = . }}{{ break }}{{ end }}{{ end }} {{- end }} {{- if $out }}{{ break }}{{ end }} {{- end }} {{- if $out }}{{ break }}{{ end }} {{- end }} {{- return $out -}}
