{{- /* On This Day's day files: /otd/.json, one a calendar day, listing every post published on that day in any year (its date in the blog's time zone, local-date.html), oldest first. Each post is u its path y its year t its title ("" for a micropost) x plain text: a titled post's summary, a micropost's opening p its first photo's original address (first-image.html), or "" js/on-this-day.js draws the page from them and js/sites.js the rail's picks; both put photos through the resizer themselves. Built once a build: call it as partialCached "otd-days.html" "otd". It returns the files' folder. */ -}} {{- $typeName := site.Params.contentTypeName | default "post" -}} {{- $days := newScratch -}} {{- range (where site.RegularPages "Type" $typeName).ByDate -}} {{- $title := .Title | plainify | htmlUnescape | strings.TrimSpace -}} {{- $text := .Summary | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- $text = truncate (cond (ne $title "") 200 280) "…" $text | htmlUnescape -}} {{- $date := partial "local-date.html" . -}} {{- $days.Add ($date.Format "01-02") (slice (dict "u" .RelPermalink "y" $date.Year "t" $title "x" $text "p" (partial "first-image.html" .))) -}} {{- end -}} {{- range $day, $posts := $days.Values -}} {{- (resources.FromString (printf "otd/%s.json" $day) (jsonify $posts)).Publish -}} {{- end -}} {{- return ("otd/" | relURL) -}}
