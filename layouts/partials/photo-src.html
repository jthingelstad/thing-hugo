{{- /* A smaller copy of one of Jamie's uploaded photos, for thumbnails. Usage: {{ partial "photo-src.html" (dict "src" $url "size" 200) }} micro.blog serves a fit-resized copy at https://micro.blog/photos// (a 1.5 MB original comes back as 10 KB at 200). Only the blog's own uploads go through it. params.photo_proxy sets another service, or "off" for the originals. */ -}} {{- $src := .src -}} {{- $proxy := site.Params.photo_proxy | default "https://micro.blog/photos/" -}} {{- if and (ne $proxy "off") (strings.HasPrefix $src (printf "%suploads/" site.BaseURL)) -}} {{- $src = printf "%s%d/%s" $proxy .size $src -}} {{- end -}} {{- return $src -}}
