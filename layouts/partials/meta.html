{{- /* The description and the Open Graph tags (micro.blog injects none). Takes a dict: the page, and on /archive/ and /photos/ (the home page in another output format, so its description, title and .Permalink) that page's own title, description and url, from its "meta" block. A description is 160 characters at most, where search results cut it off (Tinylytics SEO audit, 2026-10-22). truncate cuts at a word and adds " …", so 158 leaves room for it. */ -}} {{- $p := .page -}} {{- $description := .description | default "" -}} {{- if not $description -}} {{- if $p.IsHome -}} {{- $description = site.Params.subtitle | default "" | plainify | strings.TrimSpace | default (partial "site-description.html" $p) -}} {{- else -}} {{- $description = $p.Summary | plainify | htmlUnescape | strings.TrimSpace | truncate 158 -}} {{- end -}} {{- end }} {{- $title := .title | default (partial "title.html" $p) -}} {{- if and $p.IsHome (not .title) }}{{ $title = site.Title }}{{ end }} {{- $image := partial "first-image.html" $p -}} {{- if not $image }}{{ with site.Params.author.avatar }}{{ $image = . }}{{ end }}{{ end -}} {{- with $image }} {{- end }} {{- if $p.IsPage }} {{- range $p.Params.categories }} {{- end }} {{- end }}
