{{- /* The description and the Open Graph tags (micro.blog injects none). Takes a dict: the page, and on /archive/ and /photos/ (the home page in another output format, so its description, title and .Permalink) that page's own title, description and url, from its "meta" block. A description is 160 characters at most, where search results cut it off (Tinylytics SEO audit, 2026-10-22). truncate cuts at a word and adds " …", so 158 leaves room for it. */ -}} {{- $p := .page -}} {{- $description := .description | default "" -}} {{- if not $description -}} {{- if $p.IsHome -}} {{- $description = site.Params.subtitle | default "" | plainify | strings.TrimSpace | default (partial "site-description.html" $p) -}} {{- else -}} {{- $description = $p.Summary | plainify | htmlUnescape | strings.TrimSpace | truncate 158 -}} {{- end -}} {{- end }} {{- $title := .title | default (partial "title.html" $p) -}} {{- if and $p.IsHome (not .title) }}{{ $title = site.Title }}{{ end }} {{- /* og:url is the canonical: a list's later pages are their own address (/page/2/ and on), as baseof's canonical says. */ -}} {{- $url := .url -}} {{- if not $url -}} {{- with partial "pager.html" $p }}{{ $url = .URL | absURL }}{{ else }}{{ $url = $p.Permalink }}{{ end -}} {{- end }} {{- /* A post's share image is its card from cards.thingelstad.com (drawn from the post: its words, date and categories, and its first photo). micro.blog gives Hugo no last-edit date, so ?v= is a hash of what the card shows, and an edited post gets a new card. Anything else keeps the first photo, else the avatar. */ -}} {{- $card := "" -}} {{- if and $p.IsPage (findRE `^/\d{4}/\d{2}/\d{2}/[A-Za-z0-9_-]+\.html$` $p.RelPermalink) -}} {{- $v := substr (md5 (print $p.Title $p.Date.Unix $p.Params.categories $p.RawContent)) 0 10 -}} {{- $card = printf "https://cards.thingelstad.com/www%s.png?v=%s" (strings.TrimSuffix ".html" $p.RelPermalink) $v -}} {{- end -}} {{- $image := $card | default (partial "first-image.html" $p) -}} {{- if not $image }}{{ with site.Params.author.avatar }}{{ $image = . }}{{ end }}{{ end -}} {{- with $image }} {{- end }} {{- with $card }} {{- end }} {{- if $p.IsPage }} {{- range $p.Params.categories }} {{- end }} {{- end }}
